cmake_minimum_required(VERSION 3.16)
project(lazylean CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -g -mpopcnt -fno-stack-protector -fcf-protection=none")
set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g -fsanitize=address,undefined")
add_executable(lazylean
  src/name.cpp src/level.cpp src/expr.cpp src/loader.cpp src/kernel.cpp src/inductive.cpp src/kam.cpp src/fuse.cpp src/fix.cpp src/nbe.cpp src/main.cpp)
set(GMP_ROOT ${CMAKE_SOURCE_DIR}/third_party/gmp/usr)
if(EXISTS ${GMP_ROOT}/include/x86_64-linux-gnu/gmp.h)
  target_include_directories(lazylean PRIVATE src ${GMP_ROOT}/include ${GMP_ROOT}/include/x86_64-linux-gnu)
  target_link_directories(lazylean PRIVATE ${GMP_ROOT}/lib/x86_64-linux-gnu)
  set_target_properties(lazylean PROPERTIES BUILD_RPATH ${GMP_ROOT}/lib/x86_64-linux-gnu)
else()
  target_include_directories(lazylean PRIVATE src)
endif()
target_link_libraries(lazylean PRIVATE gmpxx gmp pthread)
